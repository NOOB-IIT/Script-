-- Update Player List Function (Alphabetical Order)
PlayerDropBtn.MouseButton1Click:Connect(function()
    PlayerScroll.Visible = not PlayerScroll.Visible
    if PlayerScroll.Visible then
        for _, v in pairs(PlayerScroll:GetChildren()) do
            if v:IsA("TextButton") then v:Destroy() end
        end
        local playerList = Players:GetPlayers()
        table.sort(playerList, function(a, b) return a.DisplayName:lower() < b.DisplayName:lower() end)
        for _, plr in ipairs(playerList) do
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, 0, 0, 30)
            btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            btn.Text = plr.DisplayName .. " (@" .. plr.Name .. ")"
            btn.TextColor3 = Color3.fromRGB(255,255,255)
            btn.ZIndex = 6
            btn.Parent = PlayerScroll
            btn.MouseButton1Click:Connect(function()
                TargetInput.Text = plr.DisplayName
                PlayerScroll.Visible = false
            end)
        end
        PlayerScroll.CanvasSize = UDim2.new(0, 0, 0, #playerList * 32)
    end
end)

TargetInput.Focused:Connect(function()
    if TargetInput.Text == "Type player Display Name" then
        TargetInput.Text = ""
    end
end)

-- Bulletproof Real-time Multi-Tag Handler & Lock System
local isPausedForTag = false
local function handleTagPenalty()
    if isPausedForTag then return end
    isPausedForTag = true
    -- Send anti-tag statement
    if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
        local textChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if textChannel then
            textChannel:SendAsync("I love Roblox]-: Because roblox tags")
        end
    else
        ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer("I love Roblox]-: Because roblox tags", "All")
    end
    -- Wait 7 seconds penalty
    task.wait(7)
    isPausedForTag = false
end

local function checkFilteredText(txt)
    if string.find(txt, "#") then
        task.spawn(handleTagPenalty)
    end
end

local function setupChannel(channel)
    channel.OnIncomingMessage = function(message)
        if message.TextSource and message.TextSource.UserId == Players.LocalPlayer.UserId then
            checkFilteredText(message.Text)
        end
    end
end

if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
    for _, channel in pairs(TextChatService.TextChannels:GetChildren()) do
        setupChannel(channel)
    end
    TextChatService.TextChannels.ChildAdded:Connect(setupChannel)
end

local ChatEvents = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
if ChatEvents and ChatEvents:FindFirstChild("OnMessageDoneFiltering") then
    ChatEvents.OnMessageDoneFiltering.OnClientEvent:Connect(function(packet)
        if packet.FromSpeaker == Players.LocalPlayer.Name or packet.FromSpeaker == Players.LocalPlayer.DisplayName then
            checkFilteredText(packet.Message)
        end
    end)
end

-- Chat Function Setup with Pause Locking
local function sendChatMessage(msg)
    -- Agar tag ki wajah se pause chal raha hai, toh naya message bhejne se pehle script yahin ruk jayegi
    while isPausedForTag do
        task.wait(0.2)
    end
    if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
        local textChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if textChannel then
            textChannel:SendAsync(msg)
        end
    else
        ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All")
    end
end

local isSpamming = false
local spamRoutine = nil

StartBtn.MouseButton1Click:Connect(function()
    if not keyVerified then return end
    if isSpamming then return end
    local targetName = TargetInput.Text
    local selectedCat = CategoryInput.Text
    local delayTime = tonumber(DelayInput.Text) or 2
    local durationTime = tonumber(DurationInput.Text) or 999999

    if targetName == "" or targetName == "Type player Display Name" then return end
    if selectedCat == "" or selectedCat == "Choose category" then return end

    isSpamming = true
    StartBtn.Text = "SPAMMING..."

    local spamArray = {}
    if selectedCat == "Server Players" then
        for _, p in pairs(Players:GetPlayers()) do
            table.insert(spamArray, p.DisplayName)
        end
    else
        spamArray = Categories[selectedCat]
    end

    spamRoutine = task.spawn(function()
        local startTime = tick()
        local index = 1
        while isSpamming and (tick() - startTime) < durationTime do
            local currentItem = spamArray[index]
            if not currentItem then
                index = 1
                currentItem = spamArray[index]
            end
            local message = "Script Created by NOT_IITIAN { 🚀 } @@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@@ " .. targetName .. " TMXH MHN MAI " .. currentItem
            sendChatMessage(message)
            index = index + 1
            task.wait(delayTime)
        end
        isSpamming = false
        StartBtn.Text = "START SPAM"
    end)
end)

StopBtn.MouseButton1Click:Connect(function()
    isSpamming = false
