function RedTheme.Bind(instance, prop, key)
    if not instance then return end

    pcall(function() instance[prop] = RedTheme.Get(key) end)
end

-- Unified Touch & Mouse Draggable
local function makeDraggable(handle, targetFrame)
    local dragging, dragStart, startPos = false, nil, nil
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = targetFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            RedTween.Play(targetFrame, 0.1, {
                Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            }, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        end
    end)
end

-- Time & Greeting Formatter
function RedTelemetry.UpdateTime()
    local d = os.date("*t")
    local h, m, s = d.hour, d.min, d.sec
    local ampm = "AM"
    if h >= 12 then ampm = "PM" if h > 12 then h = h - 12 end elseif h == 0 then h = 12 end
    RedTelemetry.LocalTimeFormatted = string.format("%02d:%02d:%02d %s", h, m, s, ampm)
    local raw = d.hour
    if raw >= 4 and raw < 12 then RedTelemetry.Greeting = "Good morning"
    elseif raw >= 12 and raw < 17 then RedTelemetry.Greeting = "Good afternoon"
    elseif raw >= 17 and raw < 22 then RedTelemetry.Greeting = "Good evening"
    else RedTelemetry.Greeting = "Good night" end
end

task.spawn(function()
    if not LocalPlayer then return end
    pcall(function()
        local exactTotal = 0
        local itemsCount = 0
        local assetIds = {}

        -- Query all worn appearance assets from Roblox Web APIs
        local sApp, appInfo = pcall(function()
            return Players:GetCharacterAppearanceInfoAsync(LocalPlayer.UserId)
        end)
        if sApp and appInfo and appInfo.assets then
            for _, ast in ipairs(appInfo.assets) do
                if ast.id then assetIds[ast.id] = true end
            end
        end

        -- Also check worn character accessories
        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        for _, obj in ipairs(char:GetChildren()) do
            if obj:IsA("Accessory") then
                local h = obj:FindFirstChild("Handle")
                if h then
                    local sm = h:FindFirstChildOfClass("SpecialMesh")
                    if sm and sm.MeshId ~= "" then
                        local mId = string.match(sm.MeshId, "%d+")
                        if mId then assetIds[tonumber(mId)] = true end
                    end
                end
            end
        end

        -- Query exact catalog PriceInRobux or lowest resale price
        for aId, _ in pairs(assetIds) do
            local sProd, pInfo = pcall(function()
                return MarketplaceService:GetProductInfo(aId, Enum.InfoType.Asset)
            end)
            if sProd and pInfo then
                itemsCount = itemsCount + 1
                if pInfo.PriceInRobux and pInfo.PriceInRobux > 0 then
                    exactTotal = exactTotal + pInfo.PriceInRobux
                elseif pInfo.LowestResalePrice and pInfo.LowestResalePrice > 0 then
                    exactTotal = exactTotal + pInfo.LowestResalePrice
                end
            end
            task.wait(0.03)
        end

        RedTelemetry.ExactAvatarValue = exactTotal
        RedTelemetry.AvatarItemsCount = itemsCount
        RedTelemetry.EstimatedAvatarValue = exactTotal
        RedTelemetry.AvatarValuationStatus = "100% Exact Catalog R$"
    end)
end)

-- Realtime Social & Online Friends Scanner
function RedTelemetry.ScanServer()
    pcall(function()
        local plrs = Players:GetPlayers()
        RedTelemetry.ServerPlayerCount = #plrs
        RedTelemetry.ServerMaxPlayers = Players.MaxPlayers or 20
        local fCount, friends = 0, {}
        for _, p in ipairs(plrs) do
            if p ~= LocalPlayer then
                local s, isF = pcall(function() return LocalPlayer:IsFriendsWith(p.UserId) end)
                if s and isF then
                    fCount = fCount + 1
                    table.insert(friends, { Name = p.DisplayName, User = p.Name, Id = p.UserId })
                end
            end
        end
        RedTelemetry.FriendsInServer = fCount
        RedTelemetry.FriendList = friends
    end)
end

function RedTelemetry.ScanOnlineFriendsPresences()
    if not LocalPlayer or RedTelemetry.FriendsScanning then return end
    RedTelemetry.FriendsScanning = true
    local ok, result = pcall(function()
        local success, data = pcall(function() return LocalPlayer:GetFriendsOnlineAsync(200) end)
        if success then return data end
        return LocalPlayer:GetFriendsOnline(200)
    end)
    if ok and type(result) == "table" then
        RedTelemetry.FriendsPresences = result
        RedTelemetry.TotalFriendsCount = #result
        RedTelemetry.FriendsError = nil
    else
        RedTelemetry.FriendsError = tostring(result or "Could not fetch online friends")
    end
    RedTelemetry.FriendsScanning = false
end

        local look, right = camera.CFrame.LookVector, camera.CFrame.RightVector
        local flatLook = Vector3.new(look.X, 0, look.Z)
        if flatLook.Magnitude < 0.01 then flatLook = Vector3.new(0, 0, -1) end

        local flatRight = Vector3.new(right.X, 0, right.Z)

            pcall(function()
                controller.root.AssemblyLinearVelocity = Vector3.zero
                controller.root.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    -- Shared collision ownership prevents Noclip and Underground from restoring
    -- each other's temporary collision settings in the wrong order.
    local collisionOwners = {}
    local function acquireNoclip(token, character)
        if not character then return end
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                local record = collisionOwners[part]
                if not record then
                    record = { original = part.CanCollide, owners = {} }
                    collisionOwners[part] = record
                end
                record.owners[token] = true
                part.CanCollide = false
            end
        end
    end
    local function releaseNoclip(token)
        for part, record in pairs(collisionOwners) do
            record.owners[token] = nil
            if next(record.owners) == nil then
                if part.Parent then part.CanCollide = record.original end
                collisionOwners[part] = nil
            end
        end
    end
    local function uprightRotation(root)
        local look = root.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
        return CFrame.lookAt(Vector3.zero, flat.Unit, Vector3.new(0, 1, 0))
    end
    local function captureJoints(character)
        local joints = {}
        for _, item in ipairs(character:GetDescendants()) do
            if item:IsA("Motor6D") then
                table.insert(joints, { motor = item, original = item.Transform, current = item.Transform })
            end
        end
        return joints
    end
    local function restoreJoints(joints)
        for _, entry in ipairs(joints or {}) do
            if entry.motor.Parent then entry.motor.Transform = entry.original end
        end
    end
    local function makeMotion(root, name)
        local attachment = Instance.new("Attachment")
        attachment.Name = name .. "Attachment"
        attachment.Parent = root
        local velocity = Instance.new("LinearVelocity")
        velocity.Name = name .. "Velocity"
        velocity.Attachment0 = attachment
        velocity.RelativeTo = Enum.ActuatorRelativeTo.World
        velocity.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
        velocity.ForceLimitsEnabled = false
        velocity.MaxForce = 100000000
        velocity.VectorVelocity = Vector3.zero
        velocity.Parent = root
        local orientation = Instance.new("AlignOrientation")
        orientation.Name = name .. "Orientation"
        orientation.Attachment0 = attachment
        orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
        orientation.MaxTorque = 100000000
        orientation.MaxAngularVelocity = 25
        orientation.Responsiveness = 35
        orientation.RigidityEnabled = false
        orientation.CFrame = root.CFrame.Rotation
        orientation.Parent = root
        return { attachment = attachment, velocity = velocity, orientation = orientation,
            instances = { velocity, orientation, attachment } }
    end
    local function destroyMotion(motion)
        if not motion then return end
        for _, item in ipairs(motion.instances) do item:Destroy() end
    end
    local function walkingGait(rig, speed, delta, bend)
        local moving = math.clamp(speed / 5, 0, 1)
        rig.gaitPhase = (rig.gaitPhase or 0) + delta * math.clamp(3 + speed * 0.4, 3, 15)
        local wave = math.sin(rig.gaitPhase)
        local breath = math.sin(os.clock() * 2.5)
        bend = bend or 0
        for _, entry in ipairs(rig.joints) do
            local name = entry.motor.Name:gsub(" ", "")
            local side = name:find("Right") and 1 or -1
            local swing = wave * side * moving
            local pose = CFrame.new()
            if name == "RightHip" or name == "LeftHip" then
                pose = CFrame.Angles(math.rad(swing * 32 + bend * 28), 0, 0)
            elseif name == "RightShoulder" or name == "LeftShoulder" then
                pose = CFrame.Angles(math.rad(-swing * 28 - bend * 20), 0,
                    math.rad(side * (3 + bend * 24 + breath)))
            elseif name == "RightKnee" or name == "LeftKnee" then
                pose = CFrame.Angles(math.rad(math.max(0, -swing) * 38 + bend * 38), 0, 0)
            elseif name == "RightElbow" or name == "LeftElbow" then
                pose = CFrame.Angles(math.rad(7 + math.max(0, swing) * 16), 0, 0)
            elseif name == "RightAnkle" or name == "LeftAnkle" then
                pose = CFrame.Angles(math.rad(-swing * 10), 0, 0)
            elseif name == "Waist" or name == "RootJoint" or name == "Root" then
                pose = CFrame.Angles(math.rad(bend * 12), math.rad(wave * moving * 3), 0)
            elseif name == "Neck" then pose = CFrame.Angles(math.rad(-bend * 10 + breath), 0, 0) end
            if rig.humanoid and rig.humanoid.RigType == Enum.HumanoidRigType.R6 then
                local basis = entry.motor.C0.Rotation
                pose = basis:Inverse() * pose * basis
            end
            entry.current = entry.current:Lerp(pose, 1 - math.exp(-13 * math.clamp(delta, 0, 0.1)))
            if entry.motor.Parent then entry.motor.Transform = entry.current end
        end
    end
    local function characterRay(target, origin, direction)
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { target.Parent }
        params.IgnoreWater = true
        return world:Raycast(origin, direction, params)
    end
    local function findWall(target)
        local root = getRoot(target)
        if not root then return nil end
        local move = target.MoveDirection
        local forward = move.Magnitude > 0.05 and move.Unit or root.CFrame.LookVector
        forward = Vector3.new(forward.X, 0, forward.Z)
        if forward.Magnitude < 0.01 then forward = Vector3.new(0, 0, -1) end
        forward = forward.Unit
        local right = Vector3.new(-forward.Z, 0, forward.X)
        local best, nearest
        for _, direction in ipairs({ forward, right, right * -1, forward * -1,
            (forward + right).Unit, (forward - right).Unit,
            (forward * -1 + right).Unit, (forward * -1 - right).Unit }) do
            local hit = characterRay(target, root.Position, direction * 4)
            if hit and math.abs(hit.Normal.Y) < 0.35 then
                local distance = (hit.Position - root.Position).Magnitude
                if not nearest or distance < nearest then best, nearest = hit, distance end
            end
        end
        return best
    end
    local function clearSurfaceMode(state)
        local rig = state.surfaceRig
        if not rig then return end
        state.surfaceRig = nil
        destroyMotion(rig.motion)
        restoreJoints(rig.joints)
        if rig.root.Parent then
            local position = rig.root.Position
            if state.key == "HeadWalk" and rig.floorY and rig.humanoid.Health > 0 then
                position = Vector3.new(position.X, rig.floorY + feetOffset(rig.humanoid, rig.root) + 0.1, position.Z)
            end
            if rig.humanoid.Health > 0 then rig.root.CFrame = CFrame.new(position) * uprightRotation(rig.root) end
            rig.root.AssemblyAngularVelocity = Vector3.zero
            rig.root.AssemblyLinearVelocity = Vector3.zero
        end
        releaseNoclip(rig)
        if rig.humanoid.Parent then
            rig.humanoid.PlatformStand = rig.platformStand
            rig.humanoid.AutoRotate = rig.autoRotate
            if rig.humanoid.Health > 0 and not rig.platformStand then
                rig.humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
        end
        state.statusMessage = nil
    end
    local function ensureSurfaceMode(state, target)
        local root = getRoot(target)
        if not root or root.Anchored or target.SeatPart then return end
        if state.surfaceRig and state.surfaceRig.root ~= root then clearSurfaceMode(state) end
        if state.surfaceRig then return end
        local hit
        if state.key == "WallWalk" then
            hit = findWall(target)
            if not hit then state.statusMessage = "APPROACH A WALL"; return end
        else
            hit = characterRay(target, root.Position + Vector3.new(0, 4, 0), Vector3.new(0, -128, 0))
            if not hit or hit.Normal.Y < 0.5 then state.statusMessage = "FIND A FLOOR"; return end
        end
        local head = target.Parent:FindFirstChild("Head")
        local headDistance = head and math.abs((head.Position - root.Position):Dot(root.CFrame.UpVector)) + head.Size.Y * 0.5 or 2
        local rig = { root = root, humanoid = target, motion = makeMotion(root, "Quick" .. state.key),
            joints = captureJoints(target.Parent), platformStand = target.PlatformStand,
            autoRotate = target.AutoRotate, normal = hit.Normal, lastHit = os.clock(),
            headDistance = math.max(1.2, headDistance), floorY = hit.Position.Y }
        state.surfaceRig = rig
        acquireNoclip(rig, target.Parent)
        target.PlatformStand = true
        target.AutoRotate = false
        target:ChangeState(Enum.HumanoidStateType.Physics)
        state.statusMessage = nil
    end
    local function planarInput(target)
        local camera = world.CurrentCamera
        local look = camera and camera.CFrame.LookVector or Vector3.new(0, 0, -1)
        local forward = Vector3.new(look.X, 0, look.Z)
        if forward.Magnitude < 0.01 then forward = Vector3.new(0, 0, -1) end
        forward = forward.Unit
        local right = Vector3.new(-forward.Z, 0, forward.X)
        local f, s = target.MoveDirection:Dot(forward), target.MoveDirection:Dot(right)
        if keyDown(Enum.KeyCode.W) then f = 1 end
        if keyDown(Enum.KeyCode.S) then f = -1 end
        if keyDown(Enum.KeyCode.A) then s = -1 end
        if keyDown(Enum.KeyCode.D) then s = 1 end
        return f, s, forward, right
    end
    local function stepSurfaceMode(state, target, delta)
        if not state or not state.enabled then return end
        ensureSurfaceMode(state, target)
        local rig = state.surfaceRig
        if not rig then return end
        local root, hit = rig.root, nil
        local desired, look, up, normalVelocity
        local f, s, cameraForward, cameraRight = planarInput(target)
        if state.key == "WallWalk" then
            hit = characterRay(target, root.Position, rig.normal * -(feetOffset(target, root) + 4))
            if not hit or math.abs(hit.Normal.Y) >= 0.35 then hit = findWall(target) end
            if not hit then
                if os.clock() - rig.lastHit > 0.2 then clearSurfaceMode(state) end
                return
            end
            rig.lastHit, rig.normal = os.clock(), hit.Normal
            local wallUp = Vector3.new(0, 1, 0) - rig.normal * rig.normal.Y
            wallUp = wallUp.Unit
            local wallRight = wallUp:Cross(rig.normal).Unit
            desired = wallUp * f + wallRight * s
            look = desired.Magnitude > 0.05 and desired.Unit or wallUp
            up = rig.normal
            local distance = (root.Position - hit.Position):Dot(rig.normal)
            normalVelocity = rig.normal * math.clamp((feetOffset(target, root) - distance) * 16, -45, 45)
        else
            hit = characterRay(target, root.Position + Vector3.new(0, 4, 0), Vector3.new(0, -128, 0))
            if not hit or hit.Normal.Y < 0.5 then clearSurfaceMode(state); return end
            rig.floorY = hit.Position.Y
            desired = cameraForward * f + cameraRight * s
            look = desired.Magnitude > 0.05 and desired.Unit or cameraForward
            up = Vector3.new(0, -1, 0)
            normalVelocity = Vector3.new(0,
                math.clamp((hit.Position.Y + rig.headDistance - root.Position.Y) * 16, -55, 55), 0)
        end
        if desired.Magnitude > 1 then desired = desired.Unit end
        acquireNoclip(rig, target.Parent)
        local speed = target.WalkSpeed
        rig.motion.velocity.VectorVelocity = desired * speed + normalVelocity
        local goal = CFrame.lookAt(Vector3.zero, look, up)
        rig.motion.orientation.CFrame = rig.motion.orientation.CFrame:Lerp(goal, 1 - math.exp(-10 * math.clamp(delta, 0, 0.1)))
        walkingGait(rig, desired.Magnitude * speed, delta)
    end
    local function performWallHop()
        local state = byKey.WallHop
        local target = currentHumanoid()
        if not alive or not state or not state.enabled or not target or target.Health <= 0 or target.SeatPart then return false end
        if UserInputService:GetFocusedTextBox() then return false end
        if (byKey.GroundSink and byKey.GroundSink.ground) or (byKey.HeadWalk and byKey.HeadWalk.enabled) then return false end
        for _, key in ipairs(movementGroup) do
            if byKey[key] and byKey[key].enabled and byKey[key].kind then return false end
        end
        local now = os.clock()
        if state.lastHop and now - state.lastHop < 0.14 then return false end
        local hit = findWall(target)
        if not hit then return false end
        local root = getRoot(target)
        if not root then return false end
        state.lastHop = now
        if byKey.WallWalk and byKey.WallWalk.enabled then byKey.WallWalk.setEnabled(false, true) end
        local power = target.UseJumpPower and target.JumpPower or math.sqrt(math.max(0, 2 * world.Gravity * target.JumpHeight))
        local tangent = root.AssemblyLinearVelocity - hit.Normal * root.AssemblyLinearVelocity:Dot(hit.Normal)
        tangent = Vector3.new(tangent.X, 0, tangent.Z) * 0.35
        root.AssemblyLinearVelocity = tangent + hit.Normal * 26 + Vector3.new(0, math.max(50, power), 0)
        target:ChangeState(Enum.HumanoidStateType.Jumping)
        return true
    end

    -- Underground movement uses a free X/Z velocity controller, not a permanently
    -- anchored character. A surface-level camera subject preserves normal orbit/zoom.
    local function prepareGround(target)
        local root = getRoot(target)
        if not root or root.Anchored or target.SeatPart then return nil end
        local hit = characterRay(target, root.Position + Vector3.new(0, 4, 0), Vector3.new(0, -1000, 0))
        if not hit or hit.Normal.Y < 0.35 then return nil end
        local surfaceY = hit.Position.Y + feetOffset(target, root) + 0.1
        local surface = CFrame.new(root.Position.X, surfaceY, root.Position.Z) * uprightRotation(root)
        return { surface = surface, surfaceY = surfaceY, destination = surface + Vector3.new(0, -8, 0) }
    end
    local function updateGroundCamera(ground)
        if not ground.proxy or not ground.proxy.Parent or not ground.root.Parent then return end
        local root = ground.root
        ground.proxy.Position = Vector3.new(root.Position.X, ground.surfaceY + 1.5, root.Position.Z)
        local camera = world.CurrentCamera
        if not camera then return end
        if not ground.cameras[camera] then
            ground.cameras[camera] = { subject = camera.CameraSubject, cameraType = camera.CameraType, cframe = camera.CFrame }
        end
        camera.CameraType = Enum.CameraType.Custom
        camera.CameraSubject = ground.proxy
    end
    local function cancelGroundTween(ground)
        if ground.tweenConnection then ground.tweenConnection:Disconnect(); ground.tweenConnection = nil end
        if ground.tween then ground.tween:Cancel(); ground.tween = nil end
    end
    local function finishGround(state, returnToSurface)
        local ground = state.ground
        if not ground then return end
        state.ground = nil
        cancelGroundTween(ground)
        destroyMotion(ground.motion)
        local root, target = ground.root, ground.humanoid
        if root.Parent then
            root.Anchored = true
            if returnToSurface and target.Health > 0 then
                root.CFrame = CFrame.new(root.Position.X, ground.surfaceY, root.Position.Z) * uprightRotation(root)
            end
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            root.Anchored = ground.anchored
        end
        restoreJoints(ground.joints)
        releaseNoclip(ground)
        if target.Parent then
            target.PlatformStand = ground.platformStand
            target.AutoRotate = ground.autoRotate
            if target.Health > 0 and not ground.platformStand then target:ChangeState(Enum.HumanoidStateType.GettingUp) end
        end
        for camera, saved in pairs(ground.cameras) do
            pcall(function()
                local subject = saved.subject
                camera.CameraSubject = subject and subject.Parent and subject or target
                camera.CameraType = saved.cameraType
                if saved.cameraType == Enum.CameraType.Scriptable then camera.CFrame = saved.cframe end
            end)
        end
        if ground.proxy then ground.proxy:Destroy() end
        state.pendingGround = nil
        state.statusMessage = nil
    end
    local function beginGroundReturn(state)
        local ground = state.ground
        if not ground or ground.phase == "exiting" then return end
        if not ground.root.Parent or ground.humanoid.Health <= 0 then finishGround(state, false); return end
        cancelGroundTween(ground)
        ground.phase, ground.animStarted = "exiting", os.clock()
        state.statusMessage = "RETURNING TO SURFACE"
        ground.root.Anchored = true
        ground.motion.velocity.VectorVelocity = Vector3.zero
        ground.root.AssemblyLinearVelocity = Vector3.zero
        local destination = CFrame.new(ground.root.Position.X, ground.surfaceY, ground.root.Position.Z) * uprightRotation(ground.root)
        ground.tween = TweenService:Create(ground.root,
            TweenInfo.new(0.85, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), { CFrame = destination })
        ground.tweenConnection = ground.tween.Completed:Connect(function(playback)
            if state.ground == ground and ground.phase == "exiting" and playback == Enum.PlaybackState.Completed then
                finishGround(state, true)
            end
        end)
        ground.tween:Play()
    end
    local function applyGround(state, target)
        if state.ground then return end
        local root = getRoot(target)
        local destination = state.pendingGround or prepareGround(target)
        if not root or not destination then state.statusMessage = "NO GROUND BELOW"; return end
        local ground = { root = root, humanoid = target, surfaceY = destination.surfaceY,
            surface = destination.surface, anchored = root.Anchored, platformStand = target.PlatformStand,
            autoRotate = target.AutoRotate, cameras = {}, joints = captureJoints(target.Parent),
            phase = "entering", animStarted = os.clock(), motion = makeMotion(root, "QuickUnderground") }
        state.ground = ground
        state.pendingGround = nil
        acquireNoclip(ground, target.Parent)
        local proxy = Instance.new("Part")
        proxy.Name = "QuickUndergroundSurfaceCamera"
        proxy.Size = Vector3.new(1, 1, 1)
        proxy.Anchored = true
        proxy.CanCollide = false
        proxy.CanTouch = false
        proxy.CanQuery = false
        proxy.Transparency = 1
        proxy.CastShadow = false
        proxy.Parent = world
        ground.proxy = proxy
        updateGroundCamera(ground)
        target.PlatformStand = true
        target.AutoRotate = false
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        root.Anchored = true
        state.statusMessage = "ENTERING GROUND"
        ground.tween = TweenService:Create(root,
            TweenInfo.new(0.85, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), { CFrame = destination.destination })
        ground.tweenConnection = ground.tween.Completed:Connect(function(playback)
            if state.ground == ground and state.enabled and ground.phase == "entering" and playback == Enum.PlaybackState.Completed then
                ground.phase = "walking"
                root.Anchored = false
                target:ChangeState(Enum.HumanoidStateType.Physics)
                state.statusMessage = "UNDERGROUND / MOVE"
            end
        end)
        ground.tween:Play()
    end
    local function stepGround(state, delta, time)
        local ground = state and state.ground
        if not ground or not ground.root.Parent then return end
        acquireNoclip(ground, ground.humanoid.Parent)
        local root, target = ground.root, ground.humanoid
        local bend, speed = 0, 0
        if ground.phase == "walking" and state.enabled then
            local hit = characterRay(target,
                Vector3.new(root.Position.X, ground.surfaceY + 12, root.Position.Z), Vector3.new(0, -128, 0))
            if hit and hit.Normal.Y > 0.35 then
                local desiredY = hit.Position.Y + feetOffset(target, root) + 0.1
                ground.surfaceY = ground.surfaceY + (desiredY - ground.surfaceY) * (1 - math.exp(-8 * math.clamp(delta, 0, 0.1)))
            end
            local f, s, forward, right = planarInput(target)
            local move = forward * f + right * s
            if move.Magnitude > 1 then move = move.Unit end
            speed = move.Magnitude * target.WalkSpeed
            local correction = math.clamp((ground.surfaceY - 8 - root.Position.Y) * 16, -65, 65)
            ground.motion.velocity.VectorVelocity = move * target.WalkSpeed + Vector3.new(0, correction, 0)
            local facing = move.Magnitude > 0.05 and move.Unit or forward
            ground.motion.orientation.CFrame = ground.motion.orientation.CFrame:Lerp(
                CFrame.lookAt(Vector3.zero, facing, Vector3.new(0, 1, 0)), 1 - math.exp(-10 * math.clamp(delta, 0, 0.1)))
        else
            local progress = math.clamp((time - ground.animStarted) / 0.85, 0, 1)
            bend = math.sin(progress * math.pi)
        end
        walkingGait(ground, speed, delta, bend)
        updateGroundCamera(ground)
    end

    local performTeleport

    local function safeVoidY(value)
        local killY = finiteNumber(world.FallenPartsDestroyHeight) or -500
        return math.max(value, killY + 25)
    end

    local function restoreExtra(state, characterOnly, animateGround)
        if QuickPlus and QuickPlus.restore(state, characterOnly) then return true end
        if state.key == "ClickTP" then
            disconnectList(state.toolConnections)
            state.toolConnections = nil
            if state.tool then state.tool:Destroy(); state.tool = nil end
            state.toolMouse = nil
            state.statusMessage = nil
            if QuickPlus then QuickPlus.releaseTool(state) end
            return true
        elseif state.key == "SelfFling" or state.key == "Spin" then
            local spin = state.spin
            state.spin = nil
            if spin then
                spin.orientation:Destroy()
                spin.attachment:Destroy()
                if spin.root.Parent then
                    spin.root.AssemblyAngularVelocity = Vector3.zero
                    if spin.humanoid.Health > 0 then spin.root.CFrame = CFrame.new(spin.root.Position) * uprightRotation(spin.root) end
                end
                if spin.humanoid.Parent then
                    spin.humanoid.AutoRotate = spin.autoRotate
                    spin.humanoid.PlatformStand = spin.platformStand
                end
            end
            return true
        elseif state.key == "AntiFling" then
            for part, original in pairs(state.colliders or {}) do
                if part.Parent then part.CanCollide = original end
            end
            state.colliders = nil
            state.safeCFrame = nil
            return true
        elseif state.key == "AntiVoid" then
            if state.voidPlatform then state.voidPlatform:Destroy(); state.voidPlatform = nil end
            return true
        elseif state.key == "GroundSink" then
            if animateGround and not characterOnly and alive and state.ground then beginGroundReturn(state)
            else finishGround(state, true) end
            state.pendingGround = nil
            return true
        elseif state.key == "Noclip" then
            releaseNoclip(state)
            return true
        elseif state.key == "WallWalk" or state.key == "HeadWalk" then
            clearSurfaceMode(state)
            return true
        elseif state.key == "WallHop" then
            state.lastHop = nil
            return true
        end
        return false
    end

    local function ensureTpTool(state)
        if state.tool and state.tool.Parent then return end
        disconnectList(state.toolConnections)
        if state.tool then state.tool:Destroy() end
        local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
        if not backpack then state.statusMessage = "WAITING FOR BACKPACK"; return end
        if QuickPlus then QuickPlus.acquireTool(state) end
        local tool = Instance.new("Tool")
        tool.Name = "TP Tool"
        tool.ToolTip = "Equip, then click or tap a world surface to teleport instantly."
        tool.RequiresHandle = false
        tool.CanBeDropped = false
        tool.Enabled = true
        state.tool = tool
        state.statusMessage = "EQUIP TP TOOL"
        state.toolConnections = {
            tool.Equipped:Connect(function(mouse)
                state.toolMouse = mouse or LocalPlayer:GetMouse()
                state.statusMessage = "TOOL EQUIPPED"
                if state.repaint then state.repaint() end
            end),
            tool.Unequipped:Connect(function()
                state.toolMouse = nil
                state.statusMessage = "EQUIP TP TOOL"
                if state.repaint then state.repaint() end
            end),
            tool.Activated:Connect(function()
                if not alive or not state.enabled or tool.Parent ~= LocalPlayer.Character then return end
                local mouse = state.toolMouse or LocalPlayer:GetMouse()
                if not mouse or not mouse.Target then return end
                -- Do not teleport through this UI when a menu/button covers the pointer.
                if mouse.X and mouse.Y then
                    for _, frame in ipairs({ MainWindow, FloatingPill, MainGui:FindFirstChild("QuickVerticalControls") }) do
                        if frame.Visible then
                            local pos, size = frame.AbsolutePosition, frame.AbsoluteSize
                            if mouse.X >= pos.X and mouse.X <= pos.X + size.X and
                               mouse.Y >= pos.Y and mouse.Y <= pos.Y + size.Y then return end
                        end
                    end
                end
                local params = RaycastParams.new()
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.FilterDescendantsInstances = { LocalPlayer.Character }
                local ray = mouse.UnitRay
                local hit = ray and world:Raycast(ray.Origin, ray.Direction * 50000, params)
                if not hit then
                    hit = { Position = mouse.Hit.Position, Normal = Vector3.new(0, 1, 0) }
                end
                performTeleport(hit)
            end)
        }
        tool.Parent = backpack
    end

    -- Upright self-spin: no AngularVelocity actuator, no PlatformStand, no forced
    -- falling state. Humanoid translation and the normal Animate script remain active.
    local function applySpin(state, target, delta)
        local root = getRoot(target)
        if not root or target.SeatPart then
            if state.spin then restoreExtra(state) end
            return
        end
        if state.spin and state.spin.root ~= root then restoreExtra(state) end
        if not state.spin then
            local attachment = Instance.new("Attachment")
            attachment.Name = "QuickUprightSpinAttachment"
            attachment.Parent = root
            local orientation = Instance.new("AlignOrientation")
            orientation.Name = "QuickUprightSpin"
            orientation.Attachment0 = attachment
            orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
            orientation.MaxTorque = 100000000
            orientation.Responsiveness = 55
            orientation.RigidityEnabled = false
            orientation.CFrame = uprightRotation(root)
            orientation.Parent = root
            root.AssemblyAngularVelocity = Vector3.zero
            local previousPlatformStand = target.PlatformStand
            local currentState = target:GetState()
            if previousPlatformStand or currentState == Enum.HumanoidStateType.FallingDown or
               currentState == Enum.HumanoidStateType.Ragdoll or currentState == Enum.HumanoidStateType.Physics then
                target.PlatformStand = false
                target:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
            state.spin = { root = root, humanoid = target, autoRotate = target.AutoRotate,
                platformStand = previousPlatformStand,
                attachment = attachment, orientation = orientation, rotation = uprightRotation(root), angle = 0 }
        end
        local spin = state.spin
        if state.value <= 0 then
            spin.orientation.Enabled = false
            target.AutoRotate = spin.autoRotate
            spin.rotation, spin.angle = uprightRotation(root), 0
            return
        end
        spin.orientation.Enabled = true
        target.AutoRotate = false
        spin.angle = (spin.angle + math.rad(state.value) * math.clamp(delta or 0, 0, 0.1)) % (2 * math.pi)
        spin.orientation.MaxAngularVelocity = math.rad(state.value) * 1.2 + 2
        spin.orientation.CFrame = spin.rotation * CFrame.Angles(0, spin.angle, 0)
    end

    local function scanAntiFling(state, target)
        state.colliders = state.colliders or {}
        local root = getRoot(target)
        local nearby = {}
        for _, player in ipairs(Players:GetPlayers()) do
            local character = player ~= LocalPlayer and player.Character
            local otherRoot = character and character:FindFirstChild("HumanoidRootPart")
            if root and otherRoot and (otherRoot.Position - root.Position).Magnitude < 60 then
                for _, part in ipairs(character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        nearby[part] = true
                        if state.colliders[part] == nil then state.colliders[part] = part.CanCollide end
                        part.CanCollide = false
                    end
                end
            end
        end
        for part, original in pairs(state.colliders) do
            if not nearby[part] then
                if part.Parent then part.CanCollide = original end
                state.colliders[part] = nil
            end
        end
    end

    local function stepAntiFling(state, target, delta)
        if not state or not state.enabled then return end
        if byKey.GroundSink and byKey.GroundSink.ground then return end
        local root = getRoot(target)
        if not root or root.Anchored or target.SeatPart then return end
        state.scanElapsed = (state.scanElapsed or 1) + delta
        if state.scanElapsed >= 0.15 then state.scanElapsed = 0; scanAntiFling(state, target) end
        local limit = math.max(200, target.WalkSpeed * 2 + 100,
            (target.UseJumpPower and target.JumpPower or 50) * 2 + 100)
        for _, key in ipairs(movementGroup) do
            local mode = byKey[key]
            if mode and mode.enabled and mode.value then limit = math.max(limit, mode.value * 2 + 100) end
        end
        local linear, angular = root.AssemblyLinearVelocity, root.AssemblyAngularVelocity
        if linear.Magnitude > limit or angular.Magnitude > 50 then
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            if state.safeCFrame then root.CFrame = state.safeCFrame end
        else
            state.safeCFrame = root.CFrame
        end
    end

    local function stepAntiVoid(state, target, delta)
        if not state or not state.enabled or (byKey.GroundSink and byKey.GroundSink.ground) then return end
        local root = getRoot(target)
        if not root or root.Anchored or target.SeatPart then return end
        state.value = safeVoidY(state.value)
        if not state.voidPlatform or not state.voidPlatform.Parent then
            local platform = Instance.new("Part")
            platform.Name = "QuickAntiVoidFloor"
            platform.Size = Vector3.new(128, 1, 128)
            platform.Anchored = true
            platform.Transparency = 1
            platform.CanCollide = true
            platform.CanTouch = false
            platform.CanQuery = false
            platform.CastShadow = false
            platform.Parent = world
            state.voidPlatform = platform
        end
        local floorY = state.value - feetOffset(target, root)
        state.voidPlatform.Position = Vector3.new(root.Position.X, floorY - 0.5, root.Position.Z)
        local velocity = root.AssemblyLinearVelocity
        local dt = math.clamp(delta, 0, 0.25)
        local projectedY = root.Position.Y + velocity.Y * dt - 0.5 * world.Gravity * dt * dt
        if root.Position.Y < state.value or (velocity.Y < 0 and projectedY <= state.value) then
            root.CFrame = root.CFrame + Vector3.new(0, state.value - root.Position.Y, 0)
            root.AssemblyLinearVelocity = Vector3.new(velocity.X, math.max(0, velocity.Y), velocity.Z)
            if byKey.AntiFling then byKey.AntiFling.safeCFrame = root.CFrame end
        end
    end

    local function restoreFeature(state, characterOnly, animateGround)
        if state.key == "Gravity" then
            if not characterOnly and state.originalGravity ~= nil then
                pcall(function() world.Gravity = state.originalGravity end)
                state.originalGravity = nil
            end
            return
        end
        if restoreExtra(state, characterOnly, animateGround) then return end
        clearController(state)
        disconnectList(state.sitConnections)
        state.sitConnections = nil
        if state.platform then state.platform:Destroy(); state.platform = nil end
        state.floorY, state.accumulatedSpeed = nil, nil
        local target, original = state.originalHumanoid, state.original
        state.originalHumanoid, state.original = nil, nil
        state.statusMessage = nil
        if not target or not target.Parent or not original then return end
        pcall(function()
            if state.key == "WalkSpeed" or state.key == "AcceleratingWalk" then
                target.WalkSpeed = original.WalkSpeed
            elseif state.key == "JumpPower" then
                target.JumpPower = original.JumpPower
                target.UseJumpPower = original.UseJumpPower
            elseif state.key == "AntiSit" then
                target:SetStateEnabled(Enum.HumanoidStateType.Seated, original.SeatedEnabled)
            end
        end)
    end

    local function enableAntiSit(state, target)
        if state.originalHumanoid == target and state.sitConnections then return end
        restoreFeature(state)
        state.originalHumanoid = target
        state.original = { SeatedEnabled = target:GetStateEnabled(Enum.HumanoidStateType.Seated) }
        local recovering = false
        local function rejectSeat()
            if recovering or not alive or not state.enabled or not target.Parent or target.Health <= 0 then return end
            recovering = true
            target:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
            if target.Sit or target.SeatPart then
                target.Sit = false
                target:ChangeState(Enum.HumanoidStateType.GettingUp)
            end
            recovering = false
        end
        state.sitConnections = {
            target:GetPropertyChangedSignal("Sit"):Connect(rejectSeat),
            target.Seated:Connect(rejectSeat),
            target.StateChanged:Connect(function(_, newState)
                if newState == Enum.HumanoidStateType.Seated then rejectSeat() end
            end)
        }
        rejectSeat()
    end

    local function createPlatform(state, target)
        if state.platform and state.platform.Parent then return end
        local root = getRoot(target)
        if not root then return end
        state.floorY = root.Position.Y - feetOffset(target, root)
        local platform = Instance.new("Part")
        platform.Name = "DeathNote_Quick_" .. state.key
        platform.Size = Vector3.new(48, 1, 48)
        platform.Anchored = true
        platform.Transparency = 1
        platform.CanCollide = true
        platform.CanTouch = false
        platform.CanQuery = false
        platform.CastShadow = false
        platform.Position = Vector3.new(root.Position.X, state.floorY - 0.5, root.Position.Z)
        platform.Parent = world
        state.platform = platform
    end

    local function createController(state, target)
        if state.rig and target.RigType ~= state.rig then
            clearController(state)
            state.statusMessage = "REQUIRES " .. (state.rig == Enum.HumanoidRigType.R6 and "R6" or "R15")
            return
        end
        local vehicle = state.kind == "vehicle"
        local seat = vehicle and target.SeatPart or nil
        local root = vehicle and seat and (seat.AssemblyRootPart or seat) or (not vehicle and getRoot(target))
        if not root or not root.Parent or not root:IsA("BasePart") then
            clearController(state)
            state.statusMessage = vehicle and "SIT IN A VEHICLE" or "WAITING FOR ROOT"
            return
        end
        if root.Anchored then
            clearController(state)
            state.statusMessage = "ASSEMBLY IS ANCHORED"
            return
        end
        local old = state.controller
        if old and old.root == root and old.humanoid == target and old.seat == seat then
            state.statusMessage = nil
            return
        end
        clearController(state)
        local controller = { root = root, humanoid = target, seat = seat, instances = {}, joints = {},
            started = os.clock(), cruise = 0, smoothedVelocity = Vector3.zero }
        state.controller = controller
        local attachment = Instance.new("Attachment")
        attachment.Name = "QuickMovementAttachment"
        attachment.Parent = root
        table.insert(controller.instances, attachment)
        local velocity = Instance.new("LinearVelocity")
        velocity.Name = "QuickMovementVelocity"
        velocity.Attachment0 = attachment
        velocity.RelativeTo = Enum.ActuatorRelativeTo.World
        velocity.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
        velocity.MaxForce = 100000000
        velocity.ForceLimitsEnabled = false
        velocity.VectorVelocity = Vector3.zero
        velocity.Parent = root
        table.insert(controller.instances, velocity)
        local orientation = Instance.new("AlignOrientation")
        orientation.Name = "QuickMovementOrientation"
        orientation.Attachment0 = attachment
        orientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
        orientation.MaxTorque = 100000000
        orientation.Responsiveness = 25
        orientation.RigidityEnabled = false
        orientation.CFrame = root.CFrame.Rotation
        orientation.Parent = root
        table.insert(controller.instances, orientation)
        controller.velocity, controller.orientation = velocity, orientation
        if not vehicle then
            controller.flags = { PlatformStand = target.PlatformStand, AutoRotate = target.AutoRotate }
            target.AutoRotate = false
            target.PlatformStand = state.kind == "fly"
            target:ChangeState(state.kind == "swim" and Enum.HumanoidStateType.Swimming or Enum.HumanoidStateType.Physics)
            for _, item in ipairs(target.Parent:GetDescendants()) do
                if item:IsA("Motor6D") then
                    table.insert(controller.joints, { motor = item, original = item.Transform })
                end
            end
        end
        state.statusMessage = nil
    end

    -- Continuous pose animation, not a fixed statue pose. No third-party asset IDs.
    local function proceduralPose(state, time, delta)
        local controller = state.controller
        if not controller then return end
        local age = time - controller.started
        local cruise = controller.cruise or 0
        local swim = state.kind == "swim"
        local superman = state.key == "FlyR15" and state.style == "Superman"
        local ironman = state.key == "FlyR15" and state.style == "Iron Man"
        local breath = math.sin(age * 2.4)
        local wave = math.sin(age * (swim and (4 + cruise * 3) or 4.2))
        local turn = controller.turn or 0
        local alpha = 1 - math.exp(-11 * math.clamp(delta, 0, 0.1))
        for _, entry in ipairs(controller.joints) do
            local name = entry.motor.Name:gsub(" ", "")
            local side = name:find("Right") and 1 or -1
            local pose = CFrame.new()
            if name == "RightShoulder" or name == "LeftShoulder" then
                if swim then
                    pose = CFrame.Angles(math.rad(135 + wave * side * (22 + cruise * 28)),
                        math.rad(side * breath * 4), math.rad(side * 14))
                elseif superman then
                    local reach = side == 1 and (155 + cruise * 20) or (28 + cruise * 130)
                    pose = CFrame.Angles(math.rad(reach + breath * 2.5 + wave * side * 1.5),
                        math.rad(turn * 7), math.rad(side * (-7 + breath * 1.2)))
                elseif ironman then
                    pose = CFrame.Angles(math.rad(-12 - cruise * 25 + breath * 4),
                        math.rad(side * (7 + breath * 2)), math.rad(side * (18 - cruise * 8)))
                else
                    pose = CFrame.Angles(math.rad(28 + wave * side * 7), 0,
                        math.rad(side * (40 + breath * 4)))
                end
            elseif name == "RightElbow" or name == "LeftElbow" then
                local bend = swim and (25 + math.sin(age * 5 + side) * 20) or
                    (superman and (side == 1 and 5 or 20 + (1 - cruise) * 25)) or
                    (ironman and (20 - cruise * 10 + breath * 3)) or 8
                pose = CFrame.Angles(math.rad(bend), 0, 0)
            elseif name == "RightWrist" or name == "LeftWrist" then
                pose = CFrame.Angles(math.rad(ironman and (-28 + breath * 3) or -7),
                    0, math.rad(ironman and side * 10 or 0))
            elseif name == "RightHip" or name == "LeftHip" then
                local kick = swim and wave * side * (12 + cruise * 18) or
                    ((ironman and 8 or -5) * (1 - cruise) + wave * side * 2.5)
                pose = CFrame.Angles(math.rad(kick), math.rad(turn * 3), math.rad(side * (3 + breath)))
            elseif name == "RightKnee" or name == "LeftKnee" then
                local kick = swim and math.max(0, wave * side) * 35 or
                    (1 - cruise) * (side == 1 and 8 or 20) + breath * side * 3
                pose = CFrame.Angles(math.rad(kick), 0, 0)
            elseif name == "RightAnkle" or name == "LeftAnkle" then
                pose = CFrame.Angles(math.rad(-8 + cruise * 12 + breath * 2), 0, 0)
            elseif name == "Neck" then
                pose = CFrame.Angles(math.rad((swim or superman) and (-10 - cruise * 12 + breath * 2) or breath * 2),
                    math.rad(turn * -6), 0)
            elseif name == "Waist" then
                pose = CFrame.Angles(math.rad(-cruise * 5 + breath * 1.8),
                    math.rad(turn * 5), math.rad(swim and wave * 3 or breath))
            elseif name == "RootJoint" or name == "Root" then
                pose = CFrame.new(0, breath * 0.055 * (1 - cruise), 0) *
                    CFrame.Angles(math.rad(breath * 1.2), 0, math.rad(turn * 2))
            end
            entry.current = (entry.current or entry.original):Lerp(pose, alpha)
            if entry.motor.Parent then entry.motor.Transform = entry.current end
        end
    end

    local function applyFeature(state)
        if not alive or not state.enabled then return end
        if QuickPlus and QuickPlus.apply(state, currentHumanoid()) then return end
        if state.key == "Gravity" then
            if state.originalGravity == nil then state.originalGravity = world.Gravity end
            world.Gravity = state.value / METRES_PER_STUD
            return
        end
        if state.key == "ClickTP" then ensureTpTool(state); return end
        local target = currentHumanoid()
        if not target or target.Health <= 0 then return end
        if state.key == "UnlimitedJump" or state.key == "WallHop" then return end
        if state.key == "Noclip" then acquireNoclip(state, target.Parent); return end
        if state.key == "WallWalk" or state.key == "HeadWalk" then ensureSurfaceMode(state, target); return end
        if state.key == "SelfFling" or state.key == "Spin" then applySpin(state, target); return end
        if state.key == "GroundSink" then applyGround(state, target); return end
        if state.key == "AntiVoid" then stepAntiVoid(state, target, 0); return end
        if state.key == "AntiFling" then
            if not state.safeCFrame then local root = getRoot(target); state.safeCFrame = root and root.CFrame end
            scanAntiFling(state, target)
            return
        end
        if state.key == "AntiSit" then enableAntiSit(state, target); return end
        if state.kind then createController(state, target); return end
        if state.key == "AntiFall" or state.key == "SkyWalk" then createPlatform(state, target); return end
        if state.originalHumanoid ~= target then
            restoreFeature(state)
            state.originalHumanoid = target
            if state.key == "WalkSpeed" or state.key == "AcceleratingWalk" then
                state.original = { WalkSpeed = target.WalkSpeed }
            else
                state.original = { JumpPower = target.JumpPower, UseJumpPower = target.UseJumpPower }
            end
        end
        if state.key == "WalkSpeed" then
            target.WalkSpeed = state.value
        elseif state.key == "JumpPower" then
            target.UseJumpPower = true
            target.JumpPower = state.value
        elseif state.key == "AcceleratingWalk" then
            if state.accumulatedSpeed == nil then state.accumulatedSpeed = target.WalkSpeed end
            target.WalkSpeed = state.accumulatedSpeed
        end
    end

    local function stepPlatform(state, target, delta)
        if not state or not state.enabled then return end
        applyFeature(state)
        local root = getRoot(target)
        if not root or not state.platform or not state.floorY then return end
        if state.key == "SkyWalk" then
            local changeY = verticalInput() * 20 * math.clamp(delta, 0, 0.25)
            state.floorY = state.floorY + changeY
            -- Translate both player and support, including when Gravity is zero.
            if changeY ~= 0 then root.CFrame = root.CFrame + Vector3.new(0, changeY, 0) end
        end
        state.platform.Position = Vector3.new(root.Position.X, state.floorY - 0.5, root.Position.Z)
        local minimumY = state.floorY + feetOffset(target, root)
        if root.Position.Y < minimumY - 0.35 then
            local velocity = root.AssemblyLinearVelocity
            root.CFrame = root.CFrame + Vector3.new(0, minimumY - root.Position.Y, 0)
            root.AssemblyLinearVelocity = Vector3.new(velocity.X, math.max(0, velocity.Y), velocity.Z)
        end
    end

    local function stepController(state, target, time, delta)
        if not state or not state.enabled then return end
        applyFeature(state)
        local controller = state.controller
        if not controller then return end
        local direction = flightDirection(target)
        if state.kind == "vehicle" and controller.seat and direction.Magnitude < 0.01 then
            local seat = controller.seat
            if seat:IsA("VehicleSeat") then
                direction = seat.CFrame.LookVector * seat.ThrottleFloat + seat.CFrame.RightVector * seat.SteerFloat
                if direction.Magnitude > 1 then direction = direction.Unit end
            end
        end
        local dt = math.clamp(delta, 0, 0.1)
        local cruiseAlpha = 1 - math.exp(-6 * dt)
        controller.cruise = controller.cruise + (direction.Magnitude - controller.cruise) * cruiseAlpha
        controller.smoothedVelocity = controller.smoothedVelocity:Lerp(direction * state.value, 1 - math.exp(-9 * dt))
        controller.velocity.VectorVelocity = controller.smoothedVelocity
        local camera = world.CurrentCamera
        local facing = camera and camera.CFrame.LookVector or controller.root.CFrame.LookVector
        local flat = Vector3.new(facing.X, 0, facing.Z)
        if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
        flat = flat.Unit
        local right = Vector3.new(-flat.Z, 0, flat.X)
        local turn = controller.lastFacing and controller.lastFacing:Dot(right) / math.max(dt, 0.008) * 0.2 or 0
        controller.turn = math.clamp(turn + direction:Dot(right) * 0.5, -1, 1)
        controller.lastFacing = flat
        local cruise, tilt = controller.cruise, 0
        if state.kind == "swim" then tilt = -65 - cruise * 17
        elseif state.key == "FlyR15" then
            tilt = state.style == "Superman" and (-18 - cruise * 54) or (-5 - cruise * 28)
        elseif state.kind == "fly" then tilt = -8 - cruise * 40 end
        local takeoff = 1 - math.exp(-(time - controller.started) * 4)
        local bank = state.kind == "vehicle" and 0 or controller.turn * -14
        local goal = CFrame.lookAt(Vector3.zero, flat) * CFrame.Angles(math.rad(tilt * takeoff), 0, math.rad(bank))
        controller.orientation.CFrame = controller.orientation.CFrame:Lerp(goal, 1 - math.exp(-8 * dt))
        if state.kind == "swim" then target:ChangeState(Enum.HumanoidStateType.Swimming) end
        if state.kind ~= "vehicle" then proceduralPose(state, time, delta) end
    end

    local function performUnlimitedJump()
        local state = byKey.UnlimitedJump
        if byKey.GroundSink and byKey.GroundSink.ground then return end
        if (byKey.HeadWalk and byKey.HeadWalk.enabled) or (byKey.WallWalk and byKey.WallWalk.enabled) then return end
        if not alive or not state or not state.enabled or UserInputService:GetFocusedTextBox() then return end
        local target = currentHumanoid()
        local root = getRoot(target)
        if not target or target.Health <= 0 or not root then return end
        for _, key in ipairs(movementGroup) do
            if byKey[key] and byKey[key].enabled and byKey[key].kind then return end
        end
        local power = target.UseJumpPower and target.JumpPower or
            math.sqrt(math.max(0, 2 * world.Gravity * target.JumpHeight))
        local boost = math.max(50, power) * 1.25
        target:ChangeState(Enum.HumanoidStateType.Jumping)
        local velocity = root.AssemblyLinearVelocity
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, math.max(velocity.Y, boost), velocity.Z)
    end

    local function stepMovement(delta)
        if not alive then return end
        local target = currentHumanoid()
        if not target or target.Health <= 0 then return end
        if QuickPlus and QuickPlus.stepPre(delta) then return end
        local accelerating = byKey.AcceleratingWalk
        if accelerating and accelerating.enabled then
            applyFeature(accelerating)
            if target.MoveDirection.Magnitude > 0.01 then
                local nextValue = (accelerating.accumulatedSpeed or target.WalkSpeed) +
                    ACCELERATION_PER_SECOND * math.clamp(delta, 0, 0.25)
                if finiteNumber(nextValue) then
                    accelerating.accumulatedSpeed = nextValue
                    target.WalkSpeed = nextValue
                end
            end
        end
        local antiSit = byKey.AntiSit
        if antiSit and antiSit.enabled then
            applyFeature(antiSit)
            target:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
            if target.Sit then target.Sit = false; target:ChangeState(Enum.HumanoidStateType.GettingUp) end
        end
        stepPlatform(byKey.AntiFall, target, delta)
        stepPlatform(byKey.SkyWalk, target, delta)
        local time = os.clock()
        if byKey.ClickTP and byKey.ClickTP.enabled then ensureTpTool(byKey.ClickTP) end
        if byKey.SelfFling and byKey.SelfFling.enabled then applySpin(byKey.SelfFling, target, delta) end
        if byKey.Spin and byKey.Spin.enabled then applySpin(byKey.Spin, target, delta) end
        -- Underground is activated by its native Tool, not by arming its toggle.
        stepGround(byKey.GroundSink, delta, time)
        if byKey.Noclip and byKey.Noclip.enabled then acquireNoclip(byKey.Noclip, target.Parent) end
        stepSurfaceMode(byKey.WallWalk, target, delta)
        stepSurfaceMode(byKey.HeadWalk, target, delta)
        stepAntiFling(byKey.AntiFling, target, delta)
        stepAntiVoid(byKey.AntiVoid, target, delta)
        for _, key in ipairs(movementGroup) do
            local state = byKey[key]
            if state and state.kind then stepController(state, target, time, delta) end
        end
        if QuickPlus then QuickPlus.stepPost() end
    end

    -- An actual Tool invokes this only when equipped and deliberately activated.
    -- No Tween or incremental movement is used for teleportation.
    performTeleport = function(hit)
        local state = byKey.ClickTP
        if not alive or not state or not state.enabled or UserInputService:GetFocusedTextBox() then return end
        local target = currentHumanoid()
        local root = getRoot(target)
        if not target or target.Health <= 0 or not root or not hit or not hit.Position then return end
        if target.SeatPart then notify("Leave the vehicle before using TP Tool."); return end
        if QuickPlus then QuickPlus.cancelExclusive() end
        for _, key in ipairs({ "SelfFling", "Spin", "WallWalk", "HeadWalk" }) do
            local other = byKey[key]
            if other and other.enabled then other.setEnabled(false, true) end
        end
        if byKey.GroundSink and byKey.GroundSink.ground then finishGround(byKey.GroundSink, true) end
        local destination = hit.Position + Vector3.new(0, feetOffset(target, root) + 0.2, 0)
        if hit.Normal and hit.Normal.Y < 0.4 then destination = destination + hit.Normal * 3 end
        local character = target.Parent
        local desiredRoot = CFrame.new(destination) * root.CFrame.Rotation
        local modelOffset = root.CFrame:ToObjectSpace(character:GetPivot())
        character:PivotTo(desiredRoot * modelOffset)
        target:Move(Vector3.zero, false)
        target:MoveTo(destination)
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.AssemblyLinearVelocity = Vector3.zero
                part.AssemblyAngularVelocity = Vector3.zero
            end
        end
        for _, key in ipairs(movementGroup) do
            local mode = byKey[key]
            if mode and mode.controller then
                mode.controller.smoothedVelocity = Vector3.zero
                mode.controller.velocity.VectorVelocity = Vector3.zero
            end
        end
        for _, key in ipairs({ "AntiFall", "SkyWalk" }) do
            local support = byKey[key]
            if support and support.enabled then restoreFeature(support); applyFeature(support) end
        end
        if byKey.AntiFling then byKey.AntiFling.safeCFrame = root.CFrame end
    end


    -- Native tools / local effects / requested third-party Dex viewing wrapper.
    -- No hidden chat remotes, key injection, or claimed server-only visibility.
    QuickPlus = (function()
        local api = {}
        local toolOwners, originalHotbar = {}, nil
        local keyboard = { W = false, A = false, S = false, D = false }

        function api.acquireTool(state)
            if toolOwners[state] then return end
            if next(toolOwners) == nil then
                pcall(function()
                    local starter = game:GetService("StarterGui")
                    originalHotbar = starter:GetCoreGuiEnabled(Enum.CoreGuiType.Backpack)
                    starter:SetCoreGuiEnabled(Enum.CoreGuiType.Backpack, true)
                end)
            end
            toolOwners[state] = true
        end
        function api.releaseTool(state)
            toolOwners[state] = nil
            if next(toolOwners) == nil and originalHotbar ~= nil then
                pcall(function() game:GetService("StarterGui"):SetCoreGuiEnabled(Enum.CoreGuiType.Backpack, originalHotbar) end)
                originalHotbar = nil
            end
        end

        local function dragPanel(handle, panel)
            local pointer, start, original
            connect(handle.InputBegan, function(event)
                if event.UserInputType == Enum.UserInputType.MouseButton1 or event.UserInputType == Enum.UserInputType.Touch then
                    pointer, start, original = event, event.Position, panel.Position
                end
            end)
            connect(UserInputService.InputChanged, function(event)
                if not pointer or not panel.Visible then return end
                local touch = pointer.UserInputType == Enum.UserInputType.Touch and event == pointer
                local mouse = pointer.UserInputType == Enum.UserInputType.MouseButton1 and event.UserInputType == Enum.UserInputType.MouseMovement
                if touch or mouse then
                    local delta = event.Position - start
                    panel.Position = UDim2.new(original.X.Scale, original.X.Offset + delta.X, original.Y.Scale, original.Y.Offset + delta.Y)
                end
            end)
            connect(UserInputService.InputEnded, function(event)
                if event == pointer then pointer = nil end
            end)
        end

        local function destroyTool(state)
            disconnectList(state.toolConnections)
            state.toolConnections = nil
            if state.tool then state.tool:Destroy(); state.tool = nil end
            api.releaseTool(state)
        end
        local function ensureTool(state, name, equipped, activated, unequipped)
            if state.tool and state.tool.Parent then return end
            destroyTool(state)
            local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
            if not backpack then state.statusMessage = "WAITING FOR BACKPACK"; return end
            api.acquireTool(state)
            local tool = Instance.new("Tool")
            tool.Name, tool.ToolTip = name, name .. " - equip to use"
            tool.RequiresHandle, tool.CanBeDropped, tool.Enabled = false, false, true
            state.tool, state.toolConnections = tool, {}
            local function add(signal, callback) table.insert(state.toolConnections, signal:Connect(callback)) end
            if equipped then add(tool.Equipped, equipped) end
            if activated then add(tool.Activated, function()
                if alive and state.enabled and tool.Parent == LocalPlayer.Character then activated() end
            end) end
            if unequipped then add(tool.Unequipped, unequipped) end
            tool.Parent = backpack
            state.statusMessage = "EQUIP " .. string.upper(name)
        end

            local frozen = state and state.frozen
            if not frozen then return end

            if frozen.root.Parent then
                frozen.root.AssemblyLinearVelocity = Vector3.zero
                frozen.root.AssemblyAngularVelocity = Vector3.zero

        local function stopMotions()
            for _, key in ipairs(movementGroup) do
                local state = byKey[key]
                if state and state.enabled then state.setEnabled(false, true) end
            end
            for _, key in ipairs({ "WallWalk", "HeadWalk", "SelfFling", "Spin", "AntiFall" }) do

                if state and state.enabled then state.setEnabled(false, true) end
            end

        local function toggleUnderground(state)
            if state.ground then
                beginGroundReturn(state)
            else
                api.cancelExclusive()
                stopMotions()
                local target = currentHumanoid()
                state.pendingGround = target and prepareGround(target)
                if not state.pendingGround then
                    state.statusMessage = "STAND OVER A FLOOR / LEAVE SEAT"
                    notify("Underground: stand over a collidable floor and leave your seat first.")
                    return
                end
                applyGround(state, target)
            end
            if state.tool then state.tool.Name = state.ground and "Return Surface" or "Underground" end
            if state.repaint then state.repaint() end
        end
        local function toggleFreeze(state)
            if state.frozen then releaseFreeze(state); return end
            if byKey.Flashback and byKey.Flashback.replay then stopReplay(byKey.Flashback) end
            stopMotions()
            local target = currentHumanoid()
            local root = getRoot(target)
            if not target or target.Health <= 0 or not root or target.SeatPart then notify("Leave the seat and use a living character."); return end
            state.frozen = { root = root, anchored = root.Anchored }
            root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = Vector3.zero, Vector3.zero
            root.Anchored = true
            state.statusMessage = "FROZEN / ACTIVATE TO RELEASE"
            if state.tool then state.tool.Name = "Unfreeze Me" end
        end
        local function stopRepeat(reason)
            repeatGeneration = repeatGeneration + 1
            repeatJob = nil
            local state = byKey.SpamTool
            if state and state.chatStatus then state.chatStatus.Text = reason or "Stopped. No more messages will be scheduled." end
        end
        local function startRepeat(state)
            local message = state.messageInput.Text
            local count = utf8 and utf8.len(message) or #message
            if not message:match("%S") or not count or count > 200 then
                state.chatStatus.Text = "Enter 1-200 characters."; return
            end
            local interval, duration = finiteNumber(state.intervalInput.Text), finiteNumber(state.durationInput.Text)
            if not interval or not duration then state.chatStatus.Text = "Enter valid interval and duration numbers."; return end
            interval, duration = math.clamp(interval, MIN_CHAT_INTERVAL, 60), math.clamp(duration, 1, MAX_CHAT_DURATION)
            state.intervalInput.Text, state.durationInput.Text = tostring(interval), tostring(duration)
            local ok, channel = pcall(function()
                local channels = game:GetService("TextChatService"):FindFirstChild("TextChannels")
                return channels and channels:FindFirstChild("RBXGeneral")
            end)
            if not ok or not channel then state.chatStatus.Text = "Public RBXGeneral chat is unavailable. Nothing sent."; return end
            stopRepeat()
            repeatJob = { channel = channel, message = message, interval = interval, duration = duration,
                started = os.clock(), nextSend = os.clock(), generation = repeatGeneration, inFlight = false, sent = 0 }
            state.chatStatus.Text = "Running in PUBLIC GAME CHAT. Press STOP to cancel."
        end

        local function enableEsp()
            if #espConnections > 0 then return end
            espGeneration = espGeneration + 1
            local generation = espGeneration
            local function watch(player)
                if player == LocalPlayer then return end
                table.insert(espConnections, player.CharacterAdded:Connect(function(character) buildEsp(player, character, generation) end))
                if player.Character then buildEsp(player, player.Character, generation) end
            end
            for _, player in ipairs(Players:GetPlayers()) do watch(player) end
            table.insert(espConnections, Players.PlayerAdded:Connect(watch))
            table.insert(espConnections, Players.PlayerRemoving:Connect(destroyEspRecord))
        end
        local function disableEsp()
            espGeneration = espGeneration + 1
            disconnectList(espConnections); espConnections = {}
            for player in pairs(espRecords) do destroyEspRecord(player) end
        end
        local function updateEsp()
            local localRoot = getRoot(currentHumanoid())
            for player, record in pairs(espRecords) do
                local root = record.character:FindFirstChild("HumanoidRootPart")
                if player.Character ~= record.character or not record.humanoid.Parent then destroyEspRecord(player)
                else
                    local distance = localRoot and root and math.floor((root.Position - localRoot.Position).Magnitude) or 0
                    local hp, maximum = math.max(0, record.humanoid.Health), math.max(1, record.humanoid.MaxHealth)
                    record.info.Text = string.format("HP %.0f/%.0f | %d studs", hp, maximum, distance)
                    record.fill.Size = UDim2.new(math.clamp(hp / maximum, 0, 1), 0, 1, 0)
                end
            end
        end
        resetHistory = function()
            history.frames, history.head, history.count, history.last = {}, 0, 0, -math.huge
            history.character = nil
        end
        local function recordHistory()
            local state = byKey.Flashback
            if state and state.replay then return end
            local target = currentHumanoid()
            local root = getRoot(target)
            if not target or target.Health <= 0 or not root then return end
            if history.character ~= target.Parent then resetHistory(); history.character = target.Parent end
            local now = os.clock()
            if now - history.last < 0.049 then return end
            history.last = now
            local joints = {}
            for _, item in ipairs(target.Parent:GetDescendants()) do
                if item:IsA("Motor6D") then joints[item] = item.Transform end
            end
            local previous = history.frames[history.head]
            local frame = { t = now, cframe = root.CFrame, joints = joints,
                cut = previous and (root.Position - previous.cframe.Position).Magnitude > 120 or false }
            history.head = history.head % history.capacity + 1
            history.frames[history.head] = frame
            history.count = math.min(history.count + 1, history.capacity)
        end
        local function historySnapshot()
            local result = {}
            for i = 1, history.count do
                local index = (history.head - history.count + i - 1) % history.capacity + 1
                table.insert(result, history.frames[index])
            end
            return result
        end
        local function availableHistory()
            if history.count < 2 then return 0 end
            local oldest = history.frames[(history.head - history.count) % history.capacity + 1]
            return math.min(60, history.frames[history.head].t - oldest.t)
        end
        stopReplay = function(state)
            local replay = state and state.replay
            if not replay then return end
            state.replay = nil
            if replay.root.Parent then
                replay.root.AssemblyLinearVelocity, replay.root.AssemblyAngularVelocity = Vector3.zero, Vector3.zero
                replay.root.Anchored = replay.anchored
            end
            restoreJoints(replay.joints)
            releaseNoclip(replay)
            if replay.humanoid.Parent then
                replay.humanoid.PlatformStand, replay.humanoid.AutoRotate = replay.platformStand, replay.autoRotate
                if replay.humanoid.Health > 0 and not replay.platformStand then replay.humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end
            end
            for _, item in ipairs(replay.effects) do item:Destroy() end
            if replay.camera then pcall(function() replay.camera.FieldOfView = replay.fieldOfView end) end
            state.statusMessage = "REWIND STOPPED / RECORDING"
            resetHistory()
        end
        startReplay = function(state)
            if not state.enabled or state.replay then return end
            local target = currentHumanoid()
            local root = getRoot(target)
            if not target or target.Health <= 0 or not root or target.SeatPart then notify("Leave the seat before rewinding."); return end
            local available = availableHistory()
            if state.value <= 0 or available < 0.15 then state.statusMessage = "RECORD MORE HISTORY FIRST"; return end
            stopMotions()
            target, root = currentHumanoid(), getRoot(currentHumanoid())
            local frames = historySnapshot()
            local replay = { root = root, humanoid = target, frames = frames, index = #frames,
                cursor = frames[#frames].t, goal = frames[#frames].t - math.min(state.value, available),
                anchored = root.Anchored, platformStand = target.PlatformStand, autoRotate = target.AutoRotate,
                joints = captureJoints(target.Parent), effects = {} }
            state.replay = replay
            acquireNoclip(replay, target.Parent)
            root.AssemblyLinearVelocity, root.AssemblyAngularVelocity = Vector3.zero, Vector3.zero
            root.Anchored, target.PlatformStand, target.AutoRotate = true, true, false
            local attachment = Instance.new("Attachment")
            attachment.Name, attachment.Position, attachment.Parent = "QuickRewindFlame", Vector3.new(0, 0.2, 1.5), root
            local emitter = Instance.new("ParticleEmitter")
            emitter.Name, emitter.Texture = "RewindFlame", "rbxasset://textures/particles/fire_main.dds"
            emitter.Color = ColorSequence.new(Color3.fromRGB(255, 180, 50), Color3.fromRGB(255, 40, 90))
            emitter.Size = NumberSequence.new({ NumberSequenceKeypoint.new(0, 1.3), NumberSequenceKeypoint.new(1, 0) })
            emitter.Lifetime, emitter.Speed, emitter.Rate = NumberRange.new(0.2, 0.5), NumberRange.new(4, 9), 70
            emitter.EmissionDirection, emitter.LightEmission = Enum.NormalId.Back, 0.9
            emitter.Parent = attachment
            table.insert(replay.effects, emitter); table.insert(replay.effects, attachment)
            local left, right = Instance.new("Attachment"), Instance.new("Attachment")
            left.Position, right.Position = Vector3.new(-0.55, 0.2, 1.2), Vector3.new(0.55, 0.2, 1.2)
            left.Parent, right.Parent = root, root
            local trail = Instance.new("Trail")
            trail.Name, trail.Attachment0, trail.Attachment1 = "QuickRewindTrail", left, right
            trail.Lifetime, trail.MinLength, trail.FaceCamera, trail.LightEmission = 0.5, 0.05, true, 1
            trail.Color = emitter.Color
            trail.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.2), NumberSequenceKeypoint.new(1, 1) })
            trail.Parent = root
            table.insert(replay.effects, trail); table.insert(replay.effects, left); table.insert(replay.effects, right)
            local tint = Instance.new("ColorCorrectionEffect")
            tint.Name, tint.TintColor, tint.Saturation = "QuickRewindTint", Color3.fromRGB(180, 215, 255), -0.2
            tint.Parent = game:GetService("Lighting")
            table.insert(replay.effects, tint)
            replay.camera = world.CurrentCamera
            if replay.camera then
                replay.fieldOfView = replay.camera.FieldOfView
                replay.camera.FieldOfView = math.clamp(replay.fieldOfView - 6, 5, 120)
            end
            state.statusMessage = string.format("REWINDING / %.1fX", state.replaySpeed)
        end
        stepReplay = function(state, delta)
            local replay = state.replay
            if not replay then return end
            if not replay.root.Parent or replay.humanoid.Health <= 0 then stopReplay(state); return end
            replay.cursor = math.max(replay.goal, replay.cursor - math.clamp(delta, 0, 0.25) * math.clamp(state.replaySpeed or 1, 0.5, 5))
            while replay.index > 1 and replay.frames[replay.index - 1].t > replay.cursor do replay.index = replay.index - 1 end
            local high = replay.frames[replay.index]
            local low = replay.frames[math.max(1, replay.index - 1)]
            local ratio = high.t > low.t and math.clamp((replay.cursor - low.t) / (high.t - low.t), 0, 1) or 0
            if high.cut then ratio = replay.cursor < high.t and 0 or 1 end
            replay.root.CFrame = low.cframe:Lerp(high.cframe, ratio)
            for motor, transform in pairs(low.joints) do
                if motor.Parent then motor.Transform = transform:Lerp(high.joints[motor] or transform, ratio) end
            end
            acquireNoclip(replay, replay.humanoid.Parent)
            state.statusMessage = string.format("REWIND %.1fS / %.1fX", replay.cursor - replay.goal, state.replaySpeed)

        local function ownsPhysics()
            if byKey.Flashback and byKey.Flashback.replay then return true end
            if byKey.FreezeTool and byKey.FreezeTool.frozen then return true end
            if byKey.GroundSink and byKey.GroundSink.ground then return true end
            for _, key in ipairs(movementGroup) do if byKey[key] and byKey[key].controller then return true end end
            return (byKey.WallWalk and byKey.WallWalk.surfaceRig) or (byKey.HeadWalk and byKey.HeadWalk.surfaceRig)
        end
        local function restoreAntiSlap(state)
            local guard = state.slapGuard
            if not guard then return end
            state.slapGuard = nil
            disconnectList(guard.connections)
            if guard.humanoid.Parent then
                for kind, enabled in pairs(guard.states) do guard.humanoid:SetStateEnabled(kind, enabled) end
            end
        end
        local function enableAntiSlap(state, target)
            if not target then return end
            if state.slapGuard and state.slapGuard.humanoid == target then return end
            restoreAntiSlap(state)
            local root = getRoot(target)
            local guard = { humanoid = target, states = {}, connections = {}, previous = root and root.AssemblyLinearVelocity or Vector3.zero }
            state.slapGuard = guard
            for _, kind in ipairs({ Enum.HumanoidStateType.FallingDown, Enum.HumanoidStateType.Ragdoll }) do
                guard.states[kind] = target:GetStateEnabled(kind)
                target:SetStateEnabled(kind, false)
            end
            local recovering = false
            local function recover()
                if recovering or not alive or not state.enabled or target.Health <= 0 or ownsPhysics() then return end
                recovering = true
                if target.PlatformStand then target.PlatformStand = false end
                local kind = target:GetState()
                if kind == Enum.HumanoidStateType.Ragdoll or kind == Enum.HumanoidStateType.FallingDown or kind == Enum.HumanoidStateType.Physics then
                    target:ChangeState(Enum.HumanoidStateType.GettingUp)
                end
                recovering = false
            end
            table.insert(guard.connections, target.StateChanged:Connect(recover))
            table.insert(guard.connections, target:GetPropertyChangedSignal("PlatformStand"):Connect(recover))
            recover()
        end
        local function stepAntiSlap(state, target)
            if not state or not state.enabled or not target then return end
            enableAntiSlap(state, target)
            if ownsPhysics() or target.SeatPart then return end
            local root = getRoot(target)
            if not root or root.Anchored then return end
            local guard, velocity = state.slapGuard, root.AssemblyLinearVelocity
            local desired = target.MoveDirection * target.WalkSpeed
            local mismatch = (Vector3.new(velocity.X, 0, velocity.Z) - desired).Magnitude
            local sudden = (velocity - guard.previous).Magnitude > math.max(40, target.WalkSpeed * 0.8)
            local recentJump = os.clock() - (state.lastJump or -math.huge) < 0.7
            local y = velocity.Y
            if not recentJump and sudden and y > 70 then y = 0 end
            if sudden and mismatch > math.max(38, target.WalkSpeed * 1.4) then
                velocity = Vector3.new(desired.X, y, desired.Z)
            elseif y ~= velocity.Y then velocity = Vector3.new(velocity.X, y, velocity.Z) end
            local spinning = (byKey.SelfFling and byKey.SelfFling.enabled and byKey.SelfFling.value > 0) or
                (byKey.Spin and byKey.Spin.enabled and byKey.Spin.value > 0)
            if not spinning and root.AssemblyAngularVelocity.Magnitude > 35 then root.AssemblyAngularVelocity = Vector3.zero end
            root.AssemblyLinearVelocity = velocity
            guard.previous = velocity
            if target.PlatformStand then target.PlatformStand = false; target:ChangeState(Enum.HumanoidStateType.GettingUp) end
        end

        function api.apply(state, target)
            if state.key == "ESP" then enableEsp(); return true end
            if state.key == "Explorer" then
                showDex(state)
                return true
            end
            if state.key == "GroundSink" then
                ensureTool(state, "Underground", nil, function() toggleUnderground(state) end)
                if state.tool then state.tool.Name = state.ground and "Return Surface" or "Underground" end
                return true
            end
            if state.key == "GotoTool" or state.key == "ChatAgeTool" then
                local function show() showPlayersTool(state) end
                local function hide() if state.playersPanel then state.playersPanel.Visible = false end end
                ensureTool(state, state.key == "GotoTool" and "Goto Players" or "Account Age", show, show, hide)
                return true
            end
            if state.key == "FreezeTool" then
                ensureTool(state, "Freeze Me", nil, function() toggleFreeze(state) end)
                return true
            end
            if state.key == "SpamTool" then
                ensureTool(state, "Text Repeat", function() showChat(state) end, function() showChat(state) end,
                    function() stopRepeat("Stopped: Text Repeat tool unequipped."); if state.chatPanel then state.chatPanel.Visible = false end end)
                return true

                    function() clearKeys(); if state.keyboardPanel then state.keyboardPanel.Visible = false end end)

            if state.key == "ESP" then if not characterOnly then disableEsp() end; return true end
            if state.key == "Explorer" then
                if not characterOnly then setDexVisible(false) end
                return true
            end
            if state.key == "GroundSink" then destroyTool(state); return false end
            if state.key == "GotoTool" or state.key == "ChatAgeTool" then
                destroyTool(state)
                state.ageGeneration = (state.ageGeneration or 0) + 1
                state.ageResult, state.selectedPlayer = nil, nil
                state.ageSendTicket, state.ageSending = nil, false
                if state.playerStatus then state.playerStatus.Text = "Select a player below." end
                state.playerListGeneration = (state.playerListGeneration or 0) + 1
                if state.playersPanel then state.playersPanel.Visible = false end
                return true
            end
            if state.key == "FreezeTool" then releaseFreeze(state); destroyTool(state); return true end
            if state.key == "SpamTool" then
                stopRepeat("Stopped."); destroyTool(state)
                if state.chatPanel then state.chatPanel.Visible = false end
                return true
            end
            if state.key == "KeyboardTool" then
                clearKeys(); destroyTool(state)
                if state.keyboardPanel then state.keyboardPanel.Visible = false end
                return true
            end
            if state.key == "Flashback" then if state.replay then stopReplay(state) end; return true end
            if state.key == "AntiSlapped" then restoreAntiSlap(state); return true end
            return false
        end

                    end)
                end
            end
            pulse = pulse + delta
            if pulse >= 0.12 then
                pulse = 0
                if byKey.ESP and byKey.ESP.enabled then updateEsp() end

                    if dexCache.bridge then byKey.Explorer.statusMessage = "FUSION DEX / OPEN"

                if target and target.Health > 0 then

                        if byKey[key] and byKey[key].enabled then api.apply(byKey[key], target) end
                    end
                end
            end
        end

    local characterConnections = {}
    local function attachCharacter(character)
        local target = character:WaitForChild("Humanoid", 10)
        if not alive or not target or LocalPlayer.Character ~= character then return end
        disconnectList(characterConnections)
        characterConnections = {}
        for _, state in ipairs(states) do restoreFeature(state, true) end
        humanoid = target
        if QuickPlus then QuickPlus.characterReset() end
        for _, state in ipairs(states) do
            if state.enabled and state.rig and state.rig ~= target.RigType then
                state.setEnabled(false)
            end
            if not state.enabled and not state.edited and
               (state.key == "WalkSpeed" or state.key == "JumpPower") then
                state.setValue(state.key == "WalkSpeed" and target.WalkSpeed or target.JumpPower, false)
            end
            applyFeature(state)
            state.repaint()
        end
        table.insert(characterConnections, target.Died:Connect(function()
            if humanoid ~= target then return end
            for _, state in ipairs(states) do restoreFeature(state, true) end
            updateMovementHud()
        end))

        for _, state in ipairs(states) do
            if state.enabled then state.repaint() end
        end
    end)

    connect(UserInputService.InputChanged, function(input)
        if not drag or not page.Visible or not MainWindow.Visible then return end
        local pointer = drag.input
        if (pointer.UserInputType == Enum.UserInputType.Touch and input == pointer) or
           (pointer.UserInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseMovement) then
            drag.slide(input.Position.X)
        end
    end)
    connect(UserInputService.InputEnded, function(input)
        for direction, pointer in pairs(heldPointers) do
            if input == pointer then verticalHeld[direction] = false; heldPointers[direction] = nil end
        end
        if not drag then return end
        if input == drag.input or
           (drag.input.UserInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseButton1) then
            stopDragging()
        end
    end)
    connect(UserInputService.WindowFocusReleased, function()
        stopDragging()
        verticalHeld.up, verticalHeld.down = false, false
        heldPointers = {}
    end)

    local function updateAnimationVisibility()
        stopDragging()
        local visible = page.Visible and MainWindow.Visible
        for _, feature in ipairs(featureCards) do
            if visible then feature.animation:Play() else feature.animation:Pause() end
        end
        if not visible then
            for _, descendant in ipairs(page:GetDescendants()) do
                if descendant:IsA("TextBox") and descendant:IsFocused() then descendant:ReleaseFocus() end
            end
        end
    end
    connect(page:GetPropertyChangedSignal("Visible"), updateAnimationVisibility)
    connect(MainWindow:GetPropertyChangedSignal("Visible"), updateAnimationVisibility)
    updateAnimationVisibility()

    connect(MainGui.Destroying, function()
        alive = false
        stopDragging()
        disconnectList(characterConnections)
        verticalHeld.up, verticalHeld.down = false, false
        for _, state in ipairs(states) do state.enabled = false; restoreFeature(state) end
        if QuickPlus then QuickPlus.shutdown() end
        for _, row in ipairs(rows) do cancelTween(row.tween) end
        for _, feature in ipairs(featureCards) do feature.cancel() end
        for _, animation in ipairs(animations) do animation:Cancel() end
        for _, connection in ipairs(connections) do connection:Disconnect() end
    end)
end)()
end

local universeCache, nameCache = {}, {}
local function positiveNumber(value)
    local n = tonumber(value)
    return n and n > 0 and n or nil
end
local function friendPlace(f)
    local id = positiveNumber(f.PlaceId or f.placeId)
    local location = f.LastLocation or f.lastLocation
    if not id and type(location) == "string" then
        id = positiveNumber(location:match("/games/(%d+)"))
    end
    return id
end
local function friendUniverse(placeId)
    if not placeId then return nil end
    if placeId == game.PlaceId then return game.GameId end
    if universeCache[placeId] then return universeCache[placeId] end
    local ok, data = pcall(function()
        return HttpService:JSONDecode(game:HttpGet("https://apis.roblox.com/universes/v1/places/" .. placeId .. "/universe"))
    end)
    local id = ok and type(data) == "table" and positiveNumber(data.universeId) or nil
    if id then universeCache[placeId] = id end
    return id
end
local function friendGameName(f, placeId, universeId)
    if placeId and nameCache[placeId] then return nameCache[placeId] end
    local name = f.GameName or f.PlaceName or f.Location or f.LastLocation or f.lastLocation
    if type(name) ~= "string" or name == "" or name == "Website" or name == "Online" or name == "Offline" or name == "In Game" or name:find("://") then name = nil end
    if not name and universeId then
        local ok, data = pcall(function()
            return HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games?universeIds=" .. universeId))
        end)
        if ok and type(data) == "table" and data.data and data.data[1] then name = data.data[1].name end
    end
    if not name and placeId then
        local ok, info = pcall(function() return MarketplaceService:GetProductInfo(placeId) end)
        if ok and info then name = info.Name end
    end
    if name and placeId then nameCache[placeId] = name end
    return name or (placeId and ("Place " .. placeId) or "Online")
end

    if game.JobId == "" then presenceState.Text="PRESENCE: NO SERVER ID"; return end
    local ok, response = pcall(requestFunction, {Url=PRESENCE_URL, Method="POST", Headers={["Content-Type"]="application/json"}, Body=HttpService:JSONEncode({userId=LocalPlayer.UserId, placeId=game.PlaceId, jobId=game.JobId})})
    if not MainGui.Parent then return end
    local status = ok and response and tonumber(response.StatusCode or response.Status)
