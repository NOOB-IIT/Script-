task.spawn(function()
    local function replaceTitle(obj)
        if (obj:IsA("TextLabel") or obj:IsA("TextButton")) and string.find(obj.Text, "FE Trolling GUI") then
            obj.Text = string.gsub(obj.Text, "FE Trolling GUI", "NOT_IITIAN")
        end
    end

    local targets = {game:GetService("CoreGui"), game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")}
    for _, parent in ipairs(targets) do
        for _, desc in ipairs(parent:GetDescendants()) do
            replaceTitle(desc)
        end
        parent.DescendantAdded:Connect(replaceTitle)
    end
end)
